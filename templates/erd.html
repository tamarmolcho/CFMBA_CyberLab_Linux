תרשים קשרי ישויות (ERD Diagram) ומבנה הנתונים

erDiagram
    ROLE ||--o{ USER : "assigns (1:N)"
    ROLE ||--|{ ROLE_PERMISSION : "contains (1:N)"
    PERMISSION ||--|{ ROLE_PERMISSION : "granted_to (1:N)"
    
    HOST ||--o{ VIRTUAL_SYSTEM : "hosts (1:N)"
    
    USER ||--o{ ENVIRONMENT : "creates (1:N)"
    USER ||--o{ ENVIRONMENT_RUN : "launches (1:N)"
    USER ||--o{ ACTION_LOG : "executes (1:N)"
    
    ENVIRONMENT ||--|{ ENVIRONMENT_SYSTEM : "includes (1:N)"
    VIRTUAL_SYSTEM ||--o{ ENVIRONMENT_SYSTEM : "assigned_to (1:N)"
    
    ENVIRONMENT ||--o{ ENVIRONMENT_RUN : "instantiates (1:N)"

    ROLE {
        int role_id PK
        string name
        string description
    }

    USER {
        int user_id PK
        string username
        string email
        string full_name
        string password_hash
        int role_id FK
        string locale
        boolean active
        datetime created_at
    }

    PERMISSION {
        int permission_id PK
        string code
        string description
    }

    ROLE_PERMISSION {
        int role_id PK,FK
        int permission_id PK,FK
    }

    HOST {
        int host_id PK
        string name
        string address
        string mode
        datetime created_at
    }

    VIRTUAL_SYSTEM {
        int system_id PK
        int host_id FK
        string uuid
        string name
        string os_type
        string state
        string ip_address
        int memory_mb
        int cpus
        int vdi_size_mb
    }

    ENVIRONMENT {
        int environment_id PK
        int owner_user_id FK
        string name
        text description
        string course_name
        datetime created_at
    }

    ENVIRONMENT_SYSTEM {
        int env_sys_id PK
        int environment_id FK
        int system_id FK
        int boot_order
        string role_label
    }

    ENVIRONMENT_RUN {
        int run_id PK
        int environment_id FK
        int started_by_user_id FK
        datetime started_at
        datetime stopped_at
        string network_mode
        string status
    }

    ACTION_LOG {
        int log_id PK
        int user_id FK
        string action_type
        string target
        text command
        int exit_code
        datetime timestamp
    }

    THREAT_DRILL_RESULT {
        int drill_id PK
        string full_name
        float reaction_time_ms
        string tier_rating
        datetime timestamp
    }
